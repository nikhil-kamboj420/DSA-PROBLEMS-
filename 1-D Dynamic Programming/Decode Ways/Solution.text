class Solution {
    public int numDecodings(String s) {
        int n = s.length();

        int next2 = 1;
        int next1 = 1;

        for (int i = n - 1; i >= 0; i--) {
            int current = 0;

            if (s.charAt(i) != '0') {
                current = next1;

                if (i + 1 < n) {
                    int num = (s.charAt(i) - '0') * 10 + (s.charAt(i + 1) - '0');

                    if (num >= 10 && num <= 26) {
                        current += next2;
                    }
                }
            }

            next2 = next1;
            next1 = current;
        }

        return next1;
    }
}
