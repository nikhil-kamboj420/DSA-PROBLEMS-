import java.util.*;

class Solution {
    public int[] getOrder(int[][] tasks) {
        int n = tasks.length;

        int[][] sorted = new int[n][3];

        for (int i = 0; i < n; i++) {
            sorted[i][0] = tasks[i][0];
            sorted[i][1] = tasks[i][1];
            sorted[i][2] = i;
        }

        Arrays.sort(sorted, (a, b) -> Integer.compare(a[0], b[0]));

        PriorityQueue<int[]> minHeap = new PriorityQueue<>(
            (a, b) -> {
                if (a[1] != b[1]) {
                    return Integer.compare(a[1], b[1]);
                }
                return Integer.compare(a[2], b[2]);
            }
        );

        int[] result = new int[n];
        int resultIndex = 0;
        int taskIndex = 0;
        long time = 0;

        while (resultIndex < n) {

            if (minHeap.isEmpty() && time < sorted[taskIndex][0]) {
                time = sorted[taskIndex][0];
            }

            while (taskIndex < n && sorted[taskIndex][0] <= time) {
                minHeap.offer(sorted[taskIndex]);
                taskIndex++;
            }

            int[] task = minHeap.poll();

            result[resultIndex++] = task[2];

            time += task[1];
        }

        return result;
    }
}
